async function criticalStock(){const products=await query('productos','id,codigo,nombre,ubicacion,activo');const stock=await query('existencias','cantidad,reservado,danado,producto_id');const totals=new Map();stock.forEach(x=>{const v=totals.get(x.producto_id)||{fisico:0,reservado:0,danado:0};v.fisico+=Number(x.cantidad);v.reservado+=Number(x.reservado||0);v.danado+=Number(x.danado||0);totals.set(x.producto_id,v)});const data=products.filter(x=>x.activo).map(x=>{const v=totals.get(x.id)||{fisico:0,reservado:0,danado:0};return{...x,...v,disponible:v.fisico-v.reservado-v.danado}}).filter(x=>x.disponible<15).sort((a,b)=>a.disponible-b.disponible);document.querySelector('#content').innerHTML=`<div class="toolbar"><div><h2>Stock crítico</h2><p>Productos activos con menos de 15 unidades disponibles</p></div></div><div class="notice notice-critical"><b>${data.length} productos requieren atención</b></div><div class="card table-wrap"><table class="data-table"><thead><tr><th>Código</th><th>Producto</th><th>Ubicación</th><th>Físico</th><th>Reservado</th><th>Disponible</th><th>Estado</th></tr></thead><tbody>${data.map(x=>`<tr><td class="code">${escapeHtml(x.codigo||'—')}</td><td><b>${escapeHtml(x.nombre)}</b></td><td>${escapeHtml(x.ubicacion||'—')}</td><td>${x.fisico.toFixed(2)}</td><td>${x.reservado.toFixed(2)}</td><td><b class="danger-text">${x.disponible.toFixed(2)}</b></td><td><span class="badge badge-danger">${x.disponible<=0?'AGOTADO':'CRÍTICO'}</span></td></tr>`).join('')}</tbody></table></div>`;}

async function documentModal(type){
  const [terceros,almacenes,productos]=await Promise.all([
    query('terceros','id,nombre_razon_social',{order:'nombre_razon_social',asc:true}),
    query('almacenes','id,nombre',{order:'nombre',asc:true}),
    query('productos','id,codigo,nombre,precio_compra,precio_venta',{eq:{activo:true},order:'nombre',asc:true})
  ]);
  const prefix={pedido_tienda:'PT',pedido_directo:'PD',proforma:'PRO',orden_salida:'OS',recepcion:'REC',transferencia:'TRA'}[type]||'DOC';
  const lines=[];
  const isTransfer=type==='transferencia', isReceipt=type==='recepcion';
  modal(`Nuevo: ${menu.find(x=>x[0]===type)?.[2]||'documento'}`,`<form id="form"><div class="form-grid"><div class="field"><label>Número</label><input name="numero" value="${prefix}-${Date.now().toString().slice(-6)}" required></div><div class="field"><label>Fecha</label><input name="fecha" type="date" value="${new Date().toISOString().slice(0,10)}" required></div><div class="field"><label>${isReceipt?'Proveedor':'Cliente / proveedor'}</label><select name="tercero_id"><option value="">Sin tercero</option>${terceros.map(x=>`<option value="${x.id}">${escapeHtml(x.nombre_razon_social)}</option>`).join('')}</select></div><div class="field"><label>${isTransfer?'Almacén origen':'Almacén'}</label><select name="almacen_origen_id">${almacenes.map(x=>`<option value="${x.id}">${escapeHtml(x.nombre)}</option>`).join('')}</select></div>${isTransfer?`<div class="field span-2"><label>Almacén destino</label><select name="almacen_destino_id">${almacenes.map(x=>`<option value="${x.id}">${escapeHtml(x.nombre)}</option>`).join('')}</select></div>`:''}<div class="field span-2"><label>Producto</label><div class="line-entry"><select id="line-product">${productos.map(x=>`<option value="${x.id}">${escapeHtml((x.codigo||'')+' · '+x.nombre)}</option>`).join('')}</select><input id="line-qty" type="number" min="0.001" step="0.001" value="1"><input id="line-price" type="number" min="0" step="0.01" value="0"><button class="btn btn-secondary" type="button" id="add-line">+ Agregar</button></div></div><div class="span-2 card table-wrap"><table class="data-table"><thead><tr><th>Código</th><th>Producto</th><th>Cantidad</th><th>Precio</th><th>Subtotal</th><th></th></tr></thead><tbody id="line-rows"><tr><td colspan="6" class="empty">Agregue los productos del documento</td></tr></tbody></table></div><div class="field"><label>Subtotal</label><input name="subtotal" id="doc-subtotal" type="number" step="0.01" value="0" readonly></div><div class="field"><label>IGV</label><input name="igv" id="doc-igv" type="number" step="0.01" value="0"></div><div class="field"><label>Total</label><input name="total" id="doc-total" type="number" step="0.01" value="0" readonly></div><div class="field"><label>Estado</label><select name="estado"><option value="borrador">Guardar borrador</option><option value="confirmado">Confirmar operación</option></select></div><div class="field span-2"><label>Observación</label><textarea name="observacion"></textarea></div></div></form>`,async()=>{
    if(!lines.length)throw new Error('Agregue al menos un producto.');
    const f=Object.fromEntries(new FormData(document.querySelector('#form')));f.tipo=type;f.creado_por=session.user.id;f.tercero_id=f.tercero_id?Number(f.tercero_id):null;f.almacen_origen_id=Number(f.almacen_origen_id)||null;f.almacen_destino_id=f.almacen_destino_id?Number(f.almacen_destino_id):null;if(isReceipt){f.almacen_destino_id=f.almacen_origen_id;f.almacen_origen_id=null;}['subtotal','igv','total'].forEach(k=>f[k]=Number(f[k]||0));
    const {data:doc,error}=await supabase.from('documentos').insert(f).select().single();if(error)throw error;
    const details=lines.map(x=>({documento_id:doc.id,producto_id:x.id,cantidad:x.qty,precio_unitario:x.price,descuento:0}));const {error:detailError}=await supabase.from('documento_detalles').insert(details);if(detailError)throw detailError;
    if(f.estado==='confirmado'){for(const x of lines){if(isReceipt)await runStock(x.id,f.almacen_destino_id,'entrada',x.qty,doc.id,`Recepción ${f.numero}`);else if(isTransfer){await runStock(x.id,f.almacen_origen_id,'salida',x.qty,doc.id,`Transferencia ${f.numero}`);await runStock(x.id,f.almacen_destino_id,'entrada',x.qty,doc.id,`Transferencia ${f.numero}`);}else if(type==='orden_salida'||type==='pedido_directo')await runStock(x.id,f.almacen_origen_id,'salida',x.qty,doc.id,`Salida ${f.numero}`);}}
    closeModal();toast('Operación guardada correctamente');documents(type);
  });
  const render=()=>{const sub=lines.reduce((a,x)=>a+x.qty*x.price,0);document.querySelector('#line-rows').innerHTML=lines.length?lines.map((x,i)=>`<tr><td class="code">${escapeHtml(x.codigo||'—')}</td><td>${escapeHtml(x.nombre)}</td><td>${x.qty}</td><td>${money(x.price)}</td><td>${money(x.qty*x.price)}</td><td><button type="button" class="icon-btn remove-line" data-i="${i}">✕</button></td></tr>`).join(''):`<tr><td colspan="6" class="empty">Agregue los productos del documento</td></tr>`;document.querySelector('#doc-subtotal').value=sub.toFixed(2);document.querySelector('#doc-total').value=(sub+Number(document.querySelector('#doc-igv').value||0)).toFixed(2);document.querySelectorAll('.remove-line').forEach(b=>b.onclick=()=>{lines.splice(Number(b.dataset.i),1);render()});};
  document.querySelector('#line-product').onchange=e=>{const p=productos.find(x=>x.id===Number(e.target.value));document.querySelector('#line-price').value=Number(isReceipt?p?.precio_compra:p?.precio_venta||0).toFixed(2)};document.querySelector('#line-product').dispatchEvent(new Event('change'));document.querySelector('#doc-igv').oninput=render;document.querySelector('#add-line').onclick=()=>{const p=productos.find(x=>x.id===Number(document.querySelector('#line-product').value)),qty=Number(document.querySelector('#line-qty').value),price=Number(document.querySelector('#line-price').value);if(!p||qty<=0)return;const old=lines.find(x=>x.id===p.id);if(old){old.qty+=qty;old.price=price}else lines.push({...p,qty,price});render();};
}
async function runStock(producto,almacen,tipo,cantidad,documento,observacion){const {error}=await supabase.rpc('registrar_movimiento_stock',{p_producto_id:producto,p_almacen_id:almacen,p_tipo:tipo,p_cantidad:cantidad,p_documento_id:documento,p_observacion:observacion});if(error)throw error;}

async function reports(){const [p,e,d]=await Promise.all([query('productos'),query('existencias','cantidad, productos(nombre,precio_compra,precio_venta,stock_critico)'),query('documentos')]);const inv=e.reduce((a,x)=>a+Number(x.cantidad)*Number(x.productos?.precio_compra||0),0),venta=e.reduce((a,x)=>a+Number(x.cantidad)*Number(x.productos?.precio_venta||0),0);document.querySelector('#content').innerHTML=`<div class="toolbar"><div><h2>Reportes gerenciales</h2><p>Resumen consolidado</p></div><button class="btn btn-secondary" onclick="window.print()">Imprimir</button></div><div class="stats"><div class="stat"><div class="stat-label">VALOR DE COSTO</div><div class="stat-value">${money(inv)}</div></div><div class="stat"><div class="stat-label">VALOR DE VENTA</div><div class="stat-value">${money(venta)}</div></div><div class="stat"><div class="stat-label">MARGEN POTENCIAL</div><div class="stat-value">${money(venta-inv)}</div></div><div class="stat"><div class="stat-label">DOCUMENTOS</div><div class="stat-value">${d.length}</div></div></div><div class="card"><div class="card-head"><h3>Productos con stock crítico</h3></div><div class="table-wrap"><table class="data-table"><thead><tr><th>Producto</th><th>Stock</th><th>Límite</th></tr></thead><tbody>${e.filter(x=>Number(x.cantidad)<Number(x.productos?.stock_critico||15)).map(x=>`<tr><td>${escapeHtml(x.productos?.nombre)}</td><td>${x.cantidad}</td><td>${x.productos?.stock_critico}</td></tr>`).join('')||`<tr><td colspan="3" class="empty">Sin productos críticos</td></tr>`}</tbody></table></div></div>`;}

function modal(title,body,onSave){closeModal();const wrap=document.createElement('div');wrap.className='modal-backdrop';wrap.id='modal';wrap.innerHTML=`<div class="modal"><div class="modal-head"><h3>${title}</h3><button class="icon-btn" id="close">✕</button></div><div class="modal-body">${body}</div><div class="modal-foot"><button class="btn btn-secondary" id="cancel">Cancelar</button><button class="btn btn-primary" id="save">Guardar</button></div></div>`;document.body.append(wrap);document.querySelector('#close').onclick=closeModal;document.querySelector('#cancel').onclick=closeModal;document.querySelector('#save').onclick=async e=>{try{e.target.disabled=true;await onSave()}catch(err){e.target.disabled=false;alert(err.message)}};}
function closeModal(){document.querySelector('#modal')?.remove();}
